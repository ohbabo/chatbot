document.addEventListener("DOMContentLoaded", () => {
    const csrfToken = document.querySelector('meta[name="csrf-token"]');
    // 채팅 관련 요소
    const chatForm = document.getElementById('chat-form');
    const chatBox = document.getElementById('chat-box');
    const userInput = document.getElementById('user-input');

    // 오디오 관련 요소
    const audioInput = document.getElementById('audio-input');
    const uploadAudioBtn = document.getElementById('upload-audio-btn');
    

    // 녹음 관련 요소
    const recordBtn = document.getElementById('record-btn');
    const stopBtn = document.getElementById('stop-btn');
    const recordingStatus = document.getElementById('recording-status');
    const audioPlayback = document.getElementById('audio-playback');
    let mediaRecorder;
    let audioChunks = [];


  

    // 페이지 로드 시 세션 초기화
    resetSession();

    


    chatForm.addEventListener("submit", async (e) => {
        e.preventDefault();
        const message = userInput.value.trim();

        if (message === "") return;

        appendMessage("user", message);
        userInput.value = "";
        userInput.disabled = true;

        try {
            const response = await fetch("/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                },
                body: JSON.stringify({ 
                    message: message,
                }), 
            });

             // 응답 상태 확인
             if (!response.ok) {
                const errorData = await response.text();
                console.error("Failed to send message:", errorData);
                appendMessage("assistant", "죄송합니다. 메시지 전송에 실패했습니다.");
                return;
            }

            const data = await response.json();

            if (data.response) {
                appendMessage("assistant", data.response);
            } else {
                appendMessage("assistant", "죄송합니다. 응답이 없습니다.");
                console.error("No response in data:", data);
            }
        } catch (error) {
            appendMessage("assistant", "죄송합니다. 서버에 문제가 발생했습니다.");
            console.error("Error:", error);
        } finally {
            userInput.disabled = false;
            userInput.focus();
        }
    });

     // 녹음 시작 버튼 클릭 시
    recordBtn.addEventListener("click", () => {
        if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
            navigator.mediaDevices.getUserMedia({ audio: true })
                .then(stream => {
                    mediaRecorder = new MediaRecorder(stream);
                    mediaRecorder.start();
                    console.log("MediaRecorder started");
                    recordingStatus.textContent = "녹음 중...";
                    recordBtn.disabled = true;
                    stopBtn.disabled = false;
                    audioChunks = [];

                    const recordingTimeout = setTimeout(() => {
                        if (mediaRecorder && mediaRecorder.state !== "inactive") {
                            mediaRecorder.stop();
                        }
                    }, 20000); // 20000ms = 20초

                    mediaRecorder.ondataavailable = event => {
                        audioChunks.push(event.data);
                        console.log("Data available:", event.data);
                    };

                    mediaRecorder.onstop = () => {
                        clearTimeout(recordingTimeout);
                        const audioBlob = new Blob(audioChunks, { type: 'audio/wav' });
                        sendAudio(audioBlob);
                        recordingStatus.textContent = "녹음 중지됨";
                        recordBtn.disabled = false;
                        stopBtn.disabled = true;

                        // 오디오 재생
                        const audioURL = URL.createObjectURL(audioBlob);
                        audioPlayback.src = audioURL;
                        audioPlayback.style.display = 'block';
                        console.log("Recording stopped and audio playback ready");
                    };
                })
                .catch(err => {
                    console.error("오디오 접근 권한이 거부되었습니다:", err);
                    alert("오디오 접근 권한을 허용해 주세요.");
                });
        } else {
            alert("이 브라우저는 오디오 녹음을 지원하지 않습니다.");
        }
    });

    // 녹음 중지 버튼 클릭 시
    stopBtn.addEventListener("click", () => {
        if (mediaRecorder && mediaRecorder.state !== "inactive") {
            mediaRecorder.stop();
        }
    });

    // 음성 업로드 버튼 클릭 시 처리
    uploadAudioBtn.addEventListener("click", async () => {
        const file = audioInput.files[0];
        if (!file) {
            alert("업로드할 음성 파일을 선택해주세요.");
            return;
        }

        // 오디오 재생
        const audioURL = URL.createObjectURL(file);
        audioPlayback.src = audioURL;
        audioPlayback.style.display = 'block';

        const formData = new FormData();
        formData.append('audio', file);

        appendMessage("user", "음성 메시지를 업로드 중입니다...");

        try {
            const response = await fetch('/upload_audio', {
                method: 'POST',
                body: formData
            });

            if (!response.ok) {
                const errorData = await response.text();
                console.error("Failed to upload audio:", errorData);
                appendMessage("assistant", "죄송합니다. 음성 업로드에 실패했습니다.");
                return;
            }

            const data = await response.json();

            if (data.transcript) {
                appendMessage("user", `${data.transcript} `);
                // 텍스트 메시지를 /chat 라우트로 전송
                await sendMessage(data.transcript);
            } else {
                appendMessage("assistant", "죄송합니다. 음성 인식에 실패했습니다.");
                console.error("Invalid response data:", data);
            }
        } catch (error) {
            appendMessage("assistant", "죄송합니다. 음성 업로드 중 오류가 발생했습니다.");
            console.error("Error:", error);
        }
    });

    // 오디오를 서버로 전송하는 함수
    async function sendAudio(audioBlob) {
        const formData = new FormData();
        formData.append('audio', audioBlob, 'recording.wav');
        //appendMessage("user", "음성 메시지를 업로드 중입니다...");

        try {
            const response = await fetch('/upload_audio', {
                method: 'POST',
                body: formData,
                credentials: 'same-origin', // 쿠키 포함
                
            });

            // 서버 응답 상태 확인
            if (!response.ok) {
                const errorData = await response.text();
                console.error("Failed to upload audio:", errorData);
                appendMessage("assistant", "죄송합니다. 음성 업로드에 실패했습니다.");
                return;
            }

            const data = await response.json();

            if (data.transcript) {
                appendMessage("user", `${data.transcript}`);
                // 텍스트 메시지를 /chat 라우트로 전송
                await sendMessage(data.transcript);
            } else {
                appendMessage("assistant", "죄송합니다. 음성 인식에 실패했습니다.");
                console.error("Invalid response data:", data);
            }
        } catch (error) {
            appendMessage("assistant", "죄송합니다. 음성 업로드 중 오류가 발생했습니다.");
            console.error("Error:", error);
        }
    }

     // 메시지를 서버로 전송하고 응답을 처리하는 함수
    async function sendMessage(message) {
        try {
            const response = await fetch("/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRFToken": csrfToken,
                },
                body: JSON.stringify({ 
                    message: message,
                    
                }), 
            });
    
            if (!response.ok) {
                const errorData = await response.json();
                console.error("Failed to send message:", errorData);
                appendMessage("assistant", "죄송합니다. 메시지 전송에 실패했습니다.");
                return;
            }
    
            const data = await response.json();
            console.log("Received chat response:", data);

            if (data.response) {
                appendMessage("assistant", data.response, [], data.message_id);
                // TTS 생성 요청
                if (data.message_id) {
                  generateTTSAndAddButton(data.message_id, data.response);
                }
            } else {
                appendMessage("assistant", "죄송합니다. 응답이 없습니다.");
                console.error("No response in data:", data);
            }
            } catch (error) {
              console.error("Error:", error);
              //appendMessage("assistant", "죄송합니다. 서버에 문제가 발생했습니다.");
            }
    }

    async function generateTTSAndPlay(messageId, text) {
        const ttsRequestData = {
          message_id: messageId,
          text: text
        };
        console.log("Sending TTS Request Data:", ttsRequestData);
        
        try {
          const ttsResponse = await fetch("/generate_tts", {
            method: "POST",
            headers: {
              "Content-Type": "application/json"
            },
            body: JSON.stringify(ttsRequestData)
          });
      
          if (!ttsResponse.ok) {
            const errorText = await ttsResponse.text();
            console.error(`TTS 생성 실패: ${ttsResponse.status} - ${errorText}`);
            return;
          }
      
          const ttsData = await ttsResponse.json();
          console.log("Received TTS Data:", ttsData);  // 추가된 로그
          const ttsUrls = ttsData.tts_url;  // 배열 형태
      
          if (ttsUrls && ttsUrls.length > 0) {
            playTTS(ttsUrls);
          } else {
            console.warn("No TTS URLs returned:", ttsData);
          }
        } catch (error) {
          console.error("TTS 변환 요청 실패:", error);
          appendMessage("assistant", "죄송합니다. 음성 생성 중 오류가 발생했습니다.");
        }
    }
    
    async function generateTTSAndAddButton(messageId, text) {
        const ttsRequestData = {
          message_id: messageId,
          text: text
        };
        console.log("Sending TTS Request Data:", ttsRequestData);
        
        try {
          const ttsResponse = await fetch("/generate_tts", {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
              "X-CSRFToken": csrfToken, // CSRF 토큰 추가
            },
            body: JSON.stringify(ttsRequestData)
          });
      
          if (!ttsResponse.ok) {
            const errorText = await ttsResponse.text();
            console.error(`TTS 생성 실패: ${ttsResponse.status} - ${errorText}`);
            appendMessage("assistant", "죄송합니다. 음성 생성에 실패했습니다.");
            return;
          }
      
          const ttsData = await ttsResponse.json();
          console.log("Received TTS Data:", ttsData);  // 추가된 로그
          const ttsUrls = ttsData.tts_url;  // 배열 형태
      
          if (ttsUrls && ttsUrls.length > 0) {
            addTTSButtonToMessage(messageId, ttsUrls);
          } else {
            console.warn("No TTS URLs returned:", ttsData);
          }
        } catch (error) {
          console.error("TTS 변환 요청 실패:", error);
          appendMessage("assistant", "죄송합니다. 음성 생성 중 오류가 발생했습니다.");
        }
      }
    
    // 특정 메시지에 TTS 재생 버튼을 추가하는 함수
    function addTTSButtonToMessage(messageId, ttsUrls) {
        console.log(`Adding TTS button to message ID: ${messageId}`);
        // 메시지 요소 찾기 (data-message-id 속성을 사용)
        const messageElements = chatBox.getElementsByClassName('message');
        for (let elem of messageElements) {
            if (elem.dataset.messageId == messageId) {
                // 기존에 버튼이 있는지 확인
                if (elem.querySelector('.tts-button')) return;
                
                // TTS 버튼 생성
                const playButton = document.createElement("button");
                playButton.classList.add('tts-button');
                playButton.innerText = "🔊 음성 재생";
                playButton.addEventListener("click", () => {
                    playTTS(ttsUrls);
                });
                elem.appendChild(playButton);
                break;
            }
        }
    }

    function playTTS(ttsUrls) {
        if (!ttsUrls || ttsUrls.length === 0) return;
      
        // 혹시 단일 URL일 수도 있으니 배열로 보정
        if (!Array.isArray(ttsUrls)) {
          ttsUrls = [ttsUrls];
        }
      
        let currentIndex = 0;
        const audio = new Audio(ttsUrls[currentIndex]);
      
        audio.play()
          .then(() => {
            console.log("TTS 재생 시작:", ttsUrls[currentIndex]);
          })
          .catch(err => {
            console.error("TTS 재생 오류:", err);
            alert("음성 재생 중 오류가 발생했습니다.");
          });
      
        // 한 청크 재생이 끝나면 다음 청크 재생
        audio.onended = () => {
          currentIndex++;
          if (currentIndex < ttsUrls.length) {
            audio.src = ttsUrls[currentIndex];
            audio.play().catch(err => {
              console.error("TTS 재생 오류:", err);
              alert("음성 재생 중 오류가 발생했습니다.");
            });
          }
        };
      }
    
    

    // 메시지를 채팅 박스에 추가하는 함수
    function appendMessage(sender, message, ttsUrls = [], messageId = null) {
        const messageElement = document.createElement("div");
        messageElement.classList.add("message", sender);
        
        // 메시지 ID를 데이터 속성으로 추가
        if (sender === "assistant" && messageId) {
            messageElement.dataset.messageId = messageId;
            console.log(`Message ID set: ${messageId}`);
        }
        
        // 'assistant' 역할을 'bot'으로 표시
        const displaySender = sender === 'assistant' ? 'bot' : sender;
        
        // 메시지 텍스트
        const messageText = document.createElement("p");
        messageText.innerText = `${displaySender}: ${message}`;
        messageElement.appendChild(messageText);
        
        // TTS 버튼은 생성하지 않음 (별도로 추가)
        
        chatBox.appendChild(messageElement);
        chatBox.scrollTop = chatBox.scrollHeight;
    }

   
    // 세션 초기화 함수
    function resetSession() {
        fetch("/reset", {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
                //"X-CSRFToken": csrfToken,
            },
        })
        .then(response => {
            if (!response.ok) {
                return response.text().then(text => { throw new Error(text) });
            }
            return response.json();
        })
        .then(data => {
            console.log(data.message);
            // 채팅 창 초기화
            chatBox.innerHTML = "";
            appendMessage("assistant", "새로운 대화를 시작하세요.");
        })
        .catch(error => {
            console.error("Error resetting session:", error);
            appendMessage("assistant", "세션 초기화에 실패했습니다.");
        });
    }
});

    }

    // === 텍스트 메시지를 서버로 전송 ===
    async function sendMessage(message) {
        try {
            const response = await fetch("/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRFToken": csrfToken,
                },
                body: JSON.stringify({ 
                    message: message,
                }), 
            });

            if (!response.ok) {
                const errorData = await response.json();
                console.error("Failed to send message:", errorData);
                appendMessage("assistant", "죄송합니다. 메시지 전송에 실패했습니다.");
                return;
            }

            const data = await response.json();
            console.log("Received chat response:", data);

            if (data.response) {
                let ttsUrl = null;
                if (data.message_id) {
                    ttsUrl = `/tts/${data.message_id}`;
                }
                appendMessage("assistant", data.response, ttsUrl, data.message_id);
            } else {
                appendMessage("assistant", "죄송합니다. 응답이 없습니다.");
                console.error("No response in data:", data);
            }
        } catch (error) {
            appendMessage("assistant", "죄송합니다. 서버에 문제가 발생했습니다.");
            console.error("Error:", error);
        }
    }

    // === TTS를 재생하는 함수 ===
    function playTTS(ttsUrl, messageElement) {
        if (!ttsUrl) return;
        const audio = new Audio(ttsUrl);
        audio.play()
            .then(() => {
                console.log("TTS 재생 시작:", ttsUrl);
            })
            .catch(err => {
                console.error("TTS 재생 오류:", err);
                // 자동재생이 막힐 수 있으므로, 재생 버튼 표시
                // messageElement는 appendMessage에서 넘긴 DOM (버튼 삽입 위치)
                const retryBtn = document.createElement("button");
                retryBtn.innerText = "🔊 다시 재생";
                retryBtn.addEventListener("click", () => {
                    audio.play().catch(playErr => {
                        console.error("수동 재생도 실패:", playErr);
                    });
                });
                messageElement.appendChild(retryBtn);
            });
    }

    // === 메시지를 채팅 박스에 추가 ===
    function appendMessage(sender, message, ttsUrl = "", messageId = null) {
        const messageElement = document.createElement("div");
        messageElement.classList.add("message", sender);
        
        if (sender === "assistant" && messageId) {
            messageElement.dataset.messageId = messageId;
        }

        const displaySender = sender === 'assistant' ? 'bot' : sender;
        const messageText = document.createElement("p");
        messageText.innerText = `${displaySender}: ${message}`;
        messageElement.appendChild(messageText);

        chatBox.appendChild(messageElement);
        chatBox.scrollTop = chatBox.scrollHeight;

        // 만약 TTS URL이 있으면 자동재생 시도
        // 실패 시 catch 블록에서 버튼을 띄워줌
        if (ttsUrl) {
            playTTS(ttsUrl, messageElement);
        }
    }

    // === 세션 초기화 ===
    function resetSession() {
        fetch("/reset", {
            method: "POST",
            headers: {
                "Content-Type": "application/json",
            },
        })
        .then(response => {
            if (!response.ok) {
                return response.text().then(text => { throw new Error(text); });
            }
            return response.json();
        })
        .then(data => {
            console.log("Session reset:", data);
            // 채팅 창 초기화
            chatBox.innerHTML = "";

            // 추천 메시지 가져오기
            const recommendation = data.recommendation || "새로운 대화를 시작하세요.";
            appendMessage("assistant", recommendation, null, data.message_id);
        })
        .catch(error => {
            console.error("Error resetting session:", error);
            appendMessage("assistant", "세션 초기화에 실패했습니다.", "", null);
        });
    }
});
